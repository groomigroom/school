const c = require('ansi-colors');

function hello(name) {
    for (let i = 0; i < 10; i++) {
        console.log(c.green(name) + '님 안녕하세요');
        console.log(c.cyan(name) + '님 안녕하세요');
    }
}

hello('김구름이');
