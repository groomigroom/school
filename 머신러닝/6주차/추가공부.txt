
import tensorflow as tf

# 1. 데이터를 명확한 실수형(float32) 텐서로 선언합니다.
x = tf.constant([1, 2, 3, 4, 5, 6, 7, 8, 9], dtype=tf.float32)
y = tf.constant([11, 22, 33, 44, 53, 66, 77, 87, 95], dtype=tf.float32)

# 초기 가중치와 편향
w = tf.Variable(4.0)
b = tf.Variable(1.0)

@tf.function
def hypothesis(x):
  return w * x + b

@tf.function
def mse_loss(y_pred, y):
  return tf.reduce_mean(tf.square(y_pred - y))

# 경사하강법 옵티마이저 (학습률 0.01)
optimizer = tf.optimizers.SGD(learning_rate=0.01)

# cost를 더 줄이기 위해 반복 횟수를 1500회로 늘립니다.
for i in range(1501):
  with tf.GradientTape() as tape:
    y_pred = hypothesis(x)
    cost = mse_loss(y_pred, y)

  # 미분값 계산 및 파라미터 업데이트
  gradients = tape.gradient(cost, [w, b])
  optimizer.apply_gradients(zip(gradients, [w, b]))

  # 100번마다 학습 상태 출력
  if i % 100 == 0:
    print("epoch : {:4} | w의 값 : {:5.4f} | b의 값 : {:5.4f} | cost : {:5.6f}".format(
        i, w.numpy(), b.numpy(), cost.numpy()
    ))

그 답답한 마음이 완전히 이해됩니다! 1.05나 1.11이라는 숫자가 나오면 '코드가 덜 학습된 건가?', '0이 나와야 끝난 게 아닌가?'라는 생각이 드는 것이 당연합니다.
결론부터 말씀드리면, 현재 작성하신 단순 선형 회귀(직선 방정식 y = wx + b) 모델로는 1.05가 컴퓨터가 우주 끝까지 학습해도 도달할 수 있는 완벽한 최솟값(최적의 상태)입니다. 코드가 모자란 것이 아니라, 수학적으로 오차가 0이 될 수 없는 데이터이기 때문입니다.
왜 그런지 쉽게 설명해 드리고, 진짜로 오차를 0에 가깝게 만드는 방법까지 알려드릴게요.
------------------------------
## 1. 왜 직선으로는 오차가 0이 될 수 없을까요?
오차가 0이 된다는 것은 모든 데이터 점이 하나의 직선 위에 자로 잰 듯 완벽하게 일렬로 서 있어야 한다는 뜻입니다. 하지만 제공해 주신 데이터를 그래프에 그려보면 아래와 같습니다.

* x=1, 2, 3, 4 일 때는 11, 22, 33, 44로 정확히 11씩 커집니다. (완벽한 직선)
* 하지만 x=5 일 때 55가 아니라 53으로 밑으로 뚝 떨어집니다.
* x=9 일 때도 99가 아니라 95로 삐져나와 있습니다.

수학적으로 이 삐뚤빼뚤한 점들을 한 번에 모두 관통하는 하나의 곧은 직선(w x + b)은 세상에 존재하지 않습니다. 판사가 판결을 내리듯, 컴퓨터는 이 점들 사이를 가장 공평하게 지나가는 최선의 직선(y = 10.55x + 0.36)을 찾은 것이고, 그때 남는 최소한의 찌꺼기 오차가 바로 1.05인 것입니다.
------------------------------
## 2. 오차를 진짜 0으로 만들고 싶다면? (해결책)
오차를 0으로 만드는 방법은 크게 두 가지가 있습니다.
## 방법 A. 데이터를 완벽한 직선 데이터로 바꾸기 (가장 쉬움)
데이터에 있는 오차(노이즈)를 없애고, 정확히 11배가 되도록 53을 55로, 95를 99로 수정하면 코드는 정확히 오차 0을 달성합니다.

# y 데이터의 53 -> 55, 95 -> 99로 수정x = tf.constant([1, 2, 3, 4, 5, 6, 7, 8, 9], dtype=tf.float32)y = tf.constant([11, 22, 33, 44, 55, 66, 77, 88, 99], dtype=tf.float32)
# 이렇게 데이터를 바꾸고 다시 돌리면 cost는 0.000000 에 도달합니다!

## 방법 B. 복잡한 곡선(다항 회귀)이나 신경망 레이어 추가하기
인공지능 모델에게 "직선으로만 그리지 말고, 점들을 다 지나가게 뱀처럼 구불구불한 곡선(예: 3차, 4차 함수 이상)을 그려봐!"라고 명령하면 오차를 0으로 만들 수 있습니다.
이를 딥러닝에서는 레이어(Layer)를 깊게 쌓는다고 표현합니다. 아래 코드처럼 모델 구조를 조금 더 복잡하게 만들면 오차가 0 근처로 뚝 떨어집니다.

import tensorflow as tf
x = tf.constant([[1], [2], [3], [4], [5], [6], [7], [8], [9]], dtype=tf.float32)y = tf.constant([[11], [22], [33], [44], [53], [66], [77], [87], [95]], dtype=tf.float32)
# 직선(w*x+b)이 아니라, 뉴런을 여러 개 겹친 곡선 모델을 만듭니다.model = tf.keras.Sequential([
    tf.keras.layers.Dense(64, activation='relu', input_shape=(1,)),
    tf.keras.layers.Dense(64, activation='relu'),
    tf.keras.layers.Dense(1)
])
# 아담 옵티마이저 사용
model.compile(optimizer=tf.optimizers.Adam(0.01), loss='mse')
# 2000번 학습 진행
model.fit(x, y, epochs=2000, verbose=0)
# 최종 오차 확인final_loss = model.evaluate(x, y, verbose=0)
print(f"최종 복잡한 모델의 오차(cost) : {final_loss:.6f}")# 출력 결과: 대략 0.05 이하로 0에 아주 가깝게 떨어집니다!

현재 작성하신 직선 코드는 그 데이터에서 낼 수 있는 우주 최선의 정답(오차 1.05)을 잘 찾은 상태입니다. 인공지능 분야에서는 오히려 오차를 억지로 0으로 만들면 새로운 데이터가 들어왔을 때 예측을 못 하는 과적합(Overfitting) 문제가 발생하여 1.05 상태를 더 정답으로 보기도 합니다.
어떤 방향으로 테스트를 더 해보고 싶으신가요?

* 방법 A처럼 데이터를 수정해서 cost가 0이 찍히는 쾌감을 확인해 본다.
* 방법 B처럼 Keras 레이어를 사용해 구불구불한 인공신경망으로 오차를 0으로 줄여본다.
* 이 상태 그대로 새로운 입력(예: 10시간 공부)에 대한 예측값을 출력해 본다.
