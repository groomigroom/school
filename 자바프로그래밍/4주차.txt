~a => 1은 0으로 0은 1로 반전 시키는 거 -> 이거는 비트 연산자이다.

>>> => a >>> 2는 2칸 오른쪽으로 이동 음수를 양수로 만듦


------------------------------------------------------------------------------------

package p20260910;

import java.util.*;

public class Test001 {

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		System.out.println("숫자 하나 입력");
		int a = 10, b = 3, d=155, e=1232212;
		int c = sc.nextInt();
		System.out.println("-a= " + -a);
		System.out.println("a+c= " + (a+c));
		System.out.println("a-b= " + (a-b));
		System.out.println("a*b= " + (a*b));
		System.out.println("a/b= " + (a/b));
		System.out.println("a%b= " + (a%b));
		System.out.println("a+d*2= " + (a+d*2));
		System.out.println("a+b+c+d+a+d+e= " + (a+b+c+d+a+d+e));
		System.out.println("a-b-c-d/a/d%e= " + (a-b-c-d/a/d%e));
	}

}

package p20260910;

public class Test001 {

	public static void main(String[] args) {
		int a = 5, b = 5;
		System.out.println("++a= " + ++a);
		System.out.println("a++= " + a++);
		System.out.println("a= " + a);
		System.out.println("a+(b++)= " + (a + b++));
		System.out.println("b*100= " + (b*100));
		System.out.println("b*c= " + (b*c));
	}

}
