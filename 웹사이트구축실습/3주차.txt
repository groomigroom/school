function hello(name) {
    for (let i = 0; i < 10; i++) {
        console.log(c.green(name) + '님 안녕하세요');
        console.log(c.cyan(name) + '님 안녕하세요');
    }
}

hello('김구름이');

function displayA() {
    console.log('A');
}

function displayB(callback) {
    setTimeout(() => {
        console.log('B');
        callback();
    }, 2000);
}

function displayC(callback) {
    setTimeout(() => {
        console.log('C');
        callback();
    }, 10000);
}

function displayD() {
    console.log('김구름');
}


displayA();

displayB(function () {
    displayC(function () {
        displayD();
    });
});
