	private int number;

	}
	
	Car (int number) {
		this.number = number;
	}

		System.out.println("자동차 이름은 " + name + "입니다.");
	}
	
	void numberIntroduce() {
		System.out.println("자동차 번호는 " + number + "다");
	}
}

	public static void main(String[] args) {
		Car ca1 = new Car("자동차");

		Car ca2 = new Car("ev5");
