package MainPrintln;

class Car {
	private String name;
	
	Car(String name) {
		this.name = name;
	}
	
	void introduce() {
		System.out.println("자동차 이름은 " + name + "입니다.");
	}
}

public class MainPrintln {

	public static void main(String[] args) {
		Car ca1 = new Car("자동차");
		ca1.introduce();
		Car ca2 = new Car("teul");
		ca2.introduce();

	}

}
